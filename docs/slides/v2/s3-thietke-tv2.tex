public interface ISegmentStrategy
{
    string Code { get; }        // campaigns.SegmentCode
    string DisplayName { get; } // shown on frm206
    bool Includes(SegmentCandidate customer,
                  DateTime referenceDate);
}

public sealed class LapsedGuestSegment : ISegmentStrategy
{
    private readonly int _lapsedDays; // appsettings
    public bool Includes(SegmentCandidate c, DateTime d)
    {
        if (c.LastCheckOut is not { } last)
            return false;            // never stayed
        return (d.Date - last.Date).Days >= _lapsedDays;
    }
}
\end{lstlisting}

Đầu vào \code{SegmentCandidate}: service gom dữ liệu 1 lần $\Rightarrow$ strategy không đọc CSDL, không N+1, test không cần CSDL.
